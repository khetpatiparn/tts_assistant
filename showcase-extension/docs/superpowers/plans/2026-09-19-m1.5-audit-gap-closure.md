## Task 1: Spike — หยุด service worker จาก Playwright ได้หรือไม่

**เป้าหมาย:** ตอบคำถามเดียว แล้วรายงาน **ห้าม commit โค้ด production ใด ๆ ใน task นี้**

ผลลัพธ์ของ task นี้กำหนดว่า Task 5 จะเขียน LC03 ระดับ browser ได้หรือต้องบันทึกเป็น `BLOCKED`

**ข้อเท็จจริงที่ตรวจแล้ว (ไม่ต้องตรวจซ้ำ):** `Worker` ของ Playwright 1.63 มีแค่ `url()`, `evaluate()`,
`evaluateHandle()`, `on('close'|'console')`, `waitForEvent('close')` — **ไม่มี method หยุด worker**
แต่ CDP protocol ที่ bundle มามี `ServiceWorker.stopWorker` และ `ServiceWorker.stopAllWorkers`
และ `context.newCDPSession()` รับ `Page`/`Frame`

**Files:**
- Create (ชั่วคราว): `tests/e2e/spike-sw-stop.spec.ts` — **ลบทิ้งก่อนจบ task**

## Task 3: WP1 E2E — วัด progress และ cancel latency จริง

**Files:**
- Create: `tests/e2e/latency.spec.ts`

**Interfaces:**
- Consumes: `RunCheckpoint.lastCommitAt` (Task 2) เข้าถึงผ่าน `chrome.runtime.sendMessage({type:'run.status'})`
  → `response.resume.checkpoint.lastCommitAt` และ `response.resume.productCount`

**นิยามการวัด (spec ไม่ได้กำหนด รอบนี้กำหนดเอง):** ทุกช่วงเวลาระหว่าง run จำนวน staged ที่**แสดงบนจอ**
ต้องตามหลังจำนวนที่ **commit จริง** ไม่เกิน 1 วินาที วัดโดยสุ่มตัวอย่างถี่ ๆ ทั้งสองค่าพร้อมกัน
เมื่อใดที่ค่าที่แสดง < ค่าจริง ถือว่ากำลังล้าหลัง และอายุของความล้าหลังคือ `now − lastCommitAt`

**กับดักที่ต้องเลี่ยง:** ห้ามวัดเฉพาะ "เวลาที่ข้อความเปลี่ยน" แล้วเรียกว่า latency — นั่นคือการวัดคาบ poll
จุดยึด `lastCommitAt` คือสิ่งที่ทำให้เป็นการวัดจริง และห้ามใช้เวลารวมของ E2E แทน latency ของ UI

- [ ] **Step 1: เขียน test**

```ts
import { test, expect } from './fixtures/extension.js'

interface Sample {
  readonly at: number
  readonly displayed: number | null
  readonly committed: number
  readonly lastCommitAt: number | null
}

test('§16 — progress แสดงภายใน 1 วินาทีหลัง commit หน้า', async ({ context, extensionId }) => {
  const page = await context.newPage()
  await page.goto(`chrome-extension://${extensionId}/dashboard/index.html`)

  await page.locator('#start-button').click()

  const samples: Sample[] = await page.evaluate(async () => {
    const out: {
      at: number
      displayed: number | null
      committed: number
      lastCommitAt: number | null
    }[] = []
    const deadline = Date.now() + 30_000

    for (;;) {
      const status = await chrome.runtime.sendMessage({ type: 'run.status' })
      const text = document.querySelector('#run-status')?.textContent ?? ''
      const match = /(\d+)\s*รายการ/.exec(text)

      const resume = status?.resume ?? null
      out.push({
        at: Date.now(),
        displayed: match?.[1] === undefined ? null : Number(match[1]),
        committed: typeof resume?.productCount === 'number' ? resume.productCount : 0,
        lastCommitAt:
          typeof resume?.checkpoint?.lastCommitAt === 'number' ? resume.checkpoint.lastCommitAt : null,
      })

      if (status?.phase === 'finished') break
      if (Date.now() > deadline) break
      // 50 ms ให้ความละเอียดเหลือเฟือเทียบงบ 1000 ms โดยไม่สร้างภาระให้ระบบที่กำลังวัด
      await new Promise((r) => setTimeout(r, 50))
    }
    return out
  })

  // ต้องได้ตัวอย่างที่เห็น commit จริงมากกว่าศูนย์ มิฉะนั้นการวัดเป็น vacuous

  expect(withCommits.length).toBeGreaterThan(0)

  // ช่วงที่จอแสดงน้อยกว่าที่ commit จริง = กำลังล้าหลัง
  const staleAges = withCommits
    .filter((s) => (s.displayed ?? 0) < s.committed)
    .map((s) => s.at - (s.lastCommitAt as number))

  const worst = staleAges.length === 0 ? 0 : Math.max(...staleAges)
  console.log(`[วัดจริง] progress lag สูงสุด = ${String(worst)} ms จาก ${String(samples.length)} ตัวอย่าง`)
  expect(worst).toBeLessThanOrEqual(1000)
})
```

// tests/e2e/helpers/network.ts
/**
 * จำแนก request ที่ Playwright เห็นว่าเป็นการออกนอกเครื่องหรือไม่ — deny-by-default
 * scheme ที่ไม่ใช่ network (chrome-extension:, data:, blob: ฯลฯ) และปลายทาง loopback เท่านั้นที่ผ่าน
 * ทุกอย่างที่เหลือถือว่า external รวม ws:/wss: ที่เคยรอดจากการกรองแบบ http(s)-only
 */
const NON_NETWORK_SCHEMES = new Set([
  'chrome-extension:',
  'chrome:',
  'devtools:',
  'data:',
  'blob:',
  'about:',
  'file:',
])

const LOOPBACK_HOSTS = new Set(['127.0.0.1', 'localhost', '::1', '[::1]'])

export function isExternalRequest(rawUrl: string): boolean {
  let url: URL
  try {
    url = new URL(rawUrl)
  } catch {
    // parse ไม่ได้ = ไม่รู้จัก ต้องถือว่าน่าสงสัยไว้ก่อน
    return true
  }
  if (NON_NETWORK_SCHEMES.has(url.protocol)) return false
  return !LOOPBACK_HOSTS.has(url.hostname)
}
